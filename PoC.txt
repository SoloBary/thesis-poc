----------------------------------------------------------------
LAYER 0 — kind (Local Kubernetes Environment)
----------------------------------------------------------------

# Install kind v0.26.0
curl -Lo ./kind https://kind.sigs.k8s.io/dl/v0.26.0/kind-linux-amd64
chmod +x ./kind
sudo mv ./kind /usr/local/bin/kind

# Install kubectl
curl -LO "https://dl.k8s.io/release/$(curl -L -s https://dl.k8s.io/release/stable.txt)/bin/linux/amd64/kubectl"
sudo install -o root -g root -m 0755 kubectl /usr/local/bin/kubectl

# Install Helm
curl https://raw.githubusercontent.com/helm/helm/main/scripts/get-helm-3 | bash

# Create kind cluster config
# Note: extraMounts are required for Falco modern_ebpf to access host kernel
cat > kind-config.yaml << 'EOF'
kind: Cluster
apiVersion: kind.x-k8s.io/v1alpha4
name: thesis-lab
nodes:
- role: control-plane
  image: kindest/node:v1.32.0
  extraMounts:
  - hostPath: /proc
    containerPath: /host/proc
    readOnly: true
  - hostPath: /sys
    containerPath: /host/sys
    readOnly: true
  - hostPath: /
    containerPath: /host
    readOnly: true
    propagation: Bidirectional
  extraPortMappings:
  - containerPort: 30000
    hostPort: 30000
    protocol: TCP
EOF

# Create cluster
kind create cluster --config kind-config.yaml

# Install metrics-server (required for kubectl top)
kubectl apply -f https://github.com/kubernetes-sigs/metrics-server/releases/latest/download/components.yaml
kubectl patch deployment metrics-server -n kube-system --type='json' \
  -p='[{"op":"add","path":"/spec/template/spec/containers/0/args/-","value":"--kubelet-insecure-tls"}]'

# Verify
kubectl get nodes
kubectl get pods -A

LAYER 1 — Trivy Operator (Pre-runtime Image Scanning)
----------------------------------------------------------------

# Add Aqua Security Helm repo
helm repo add aquasecurity https://aquasecurity.github.io/helm-charts/
helm repo update

# Install Trivy Operator (using values file from repo)
helm install trivy-operator aquasecurity/trivy-operator \
  --namespace trivy-system \
  --create-namespace \
  -f trivy/helm-values.yaml

# Verify operator is running
kubectl get pods -n trivy-system

# Wait for vulnerability reports to be generated (takes 3-5 minutes)
kubectl get vulnerabilityreports -A

# Summary table across all namespaces
kubectl get vulnerabilityreports -A -o custom-columns=\
"NAMESPACE:.metadata.namespace,\
NAME:.metadata.name,\
CRITICAL:.report.summary.criticalCount,\
HIGH:.report.summary.highCount,\
MEDIUM:.report.summary.mediumCount,\
LOW:.report.summary.lowCount"

# Inspect a specific report
kubectl describe vulnerabilityreport -n big-monolith <report-name>

# Extract reports to txt files (already done in trivy/ folder)
kubectl get vulnerabilityreports -A --no-headers | while read namespace name rest; do
  kubectl get vulnerabilityreport -n $namespace $name -o yaml > trivy/${namespace}_${name}.txt
done

# Upgrade if needed
helm upgrade trivy-operator aquasecurity/trivy-operator \
  --namespace trivy-system \
  --reuse-values

LAYER 3 — Falco + Falcosidekick (Runtime Threat Detection)
----------------------------------------------------------------

# Add Falco Helm repo
helm repo add falcosecurity https://falcosecurity.github.io/charts
helm repo update

# Install Falco
# Note 1: modern_ebpf driver uses CO-RE/BTF — no kernel headers or
#         prebuilt driver required. Works on any BTF-enabled kernel (5.8+).
# Note 2: falco-exporter is deprecated since Falco 0.38 — native metrics
#         are exposed at :8765/metrics with falcosecurity_* prefix.
# Note 3: helm-values.yaml contains all custom configuration including
#         metrics, webserver, http_output and custom rules
kubectl create namespace falco

helm install falco falcosecurity/falco \
  --namespace falco \
  -f falco/helm-values.yaml \
  -f falco/custom-rules.yaml

# Verify
kubectl rollout status daemonset/falco -n falco
kubectl get pods -n falco

# Verify native metrics endpoint
kubectl port-forward -n falco \
  pod/$(kubectl get pod -n falco -l app.kubernetes.io/name=falco \
  -o jsonpath='{.items[0].metadata.name}') 8765:8765 &
curl -s localhost:8765/metrics | grep "^falcosecurity" | head -10

# Install Falcosidekick (separate chart)
helm install falcosidekick falcosecurity/falcosidekick \
  --namespace falco \
  --set webui.enabled=true \
  --set webui.replicaCount=1 \
  --set config.minimumpriority=warning

# Access Falcosidekick Web UI
kubectl port-forward svc/falcosidekick-ui -n falco 2802:2802
# → http://localhost:2802

# View Falco logs live
kubectl logs -n falco -l app.kubernetes.io/name=falco -c falco -f

# Test Falco detection (trigger a known rule)
kubectl exec -n big-monolith \
  $(kubectl get pod -n big-monolith -o jsonpath='{.items[0].metadata.name}') \
  -- cat /etc/shadow

LAYER 4 — Prometheus + Grafana (Metrics & Visualization)
----------------------------------------------------------------

# Add Prometheus community Helm repo
helm repo add prometheus-community https://prometheus-community.github.io/helm-charts
helm repo update

# Install kube-prometheus-stack (includes Prometheus + Grafana + AlertManager)
kubectl create namespace monitoring

helm install prometheus prometheus-community/kube-prometheus-stack \
  --namespace monitoring \
  -f prometheus/helm-values.yaml

# Verify
kubectl get pods -n monitoring

# Create Falco headless Service for Prometheus scraping
kubectl apply -f - << 'EOF'
apiVersion: v1
kind: Service
metadata:
  name: falco-metrics
  namespace: falco
  labels:
    app.kubernetes.io/name: falco
spec:
  selector:
    app.kubernetes.io/name: falco
  clusterIP: None
  ports:
    - name: metrics
      port: 8765
      targetPort: 8765
      protocol: TCP
EOF

# Register Falco with Prometheus Operator via ServiceMonitor
kubectl apply -f prometheus/servicemonitors.yaml

# Access Prometheus UI
kubectl port-forward svc/prometheus-kube-prometheus-prometheus -n monitoring 9090:9090
# → http://localhost:9090

# Access Grafana UI
# Note: port 3001 used to avoid conflict with AdGuard on 3000
kubectl port-forward svc/prometheus-grafana -n monitoring 3001:80
# → http://localhost:3001
# Default credentials: admin / admin

# Import Falco Dashboard
# Grafana UI → Dashboards → New → Import → Upload JSON
# Select: grafana/falco-dashboard.json
# Set datasource: Prometheus → Import
#
# IMPORTANT: Dashboard ID 11914 from grafana.com is INCOMPATIBLE with Falco 0.38+
# (metric prefix changed from falco_* to falcosecurity_*).
# The custom dashboard in this repo uses the current metric names.

# Useful Prometheus queries for Falco metrics:
#   sum(falcosecurity_falco_rules_matches_total)
#   sum by (rule) (increase(falcosecurity_falco_rules_matches_total[5m]))
#   rate(falcosecurity_scap_n_evts_total[1m])
#   falcosecurity_falco_cpu_usage_ratio * 100

LAYER 4.5 — Loki (Log Aggregation for Falco Events)
----------------------------------------------------------------

# Note: Loki complements Prometheus by aggregating Falco's full JSON event
# logs (not just metrics counters). LogQL queries enable searching for
# specific rules, containers, or namespaces.

# Add Grafana Helm repo
helm repo add grafana https://grafana.github.io/helm-charts
helm repo update

# Install Loki Stack (Loki + Promtail for log shipping)
# Disable embedded Grafana and Prometheus — using kube-prometheus-stack instead
helm install loki grafana/loki-stack \
  --namespace monitoring \
  --set grafana.enabled=false \
  --set prometheus.enabled=false

# Upgrade Loki image to latest
helm upgrade loki grafana/loki-stack \
  --namespace monitoring \
  --reuse-values \
  --set loki.image.tag=latest

# Verify
kubectl get pods -n monitoring | grep loki

# Add Loki as data source in Grafana
# Grafana UI → Connections → Data sources → Add data source → Loki
# URL: http://loki.monitoring.svc.cluster.local:3100

# Useful LogQL queries for Falco events:
#   {namespace="falco", container="falco"} |= "Warning"
#   {namespace="falco", container="falco"} |= "Critical"
#   {namespace="falco", container="falco"} | json | priority="Warning"
#   {namespace="falco", container="falco"} |= "/etc/shadow"

LAYER 5 — Istio + Kiali (Service Mesh & Traffic Visualization)
----------------------------------------------------------------

# Download Istio 1.24.0
curl -L https://istio.io/downloadIstio | ISTIO_VERSION=1.24.0 sh -
cd istio-1.24.0/
export PATH=$PWD/bin:$PATH

# Install Istio (demo profile for thesis/lab use)
istioctl install --set profile=demo -y

# Verify control plane
kubectl rollout status deployment/istiod -n istio-system
kubectl get pods -n istio-system

# Install Kiali + supporting addons
kubectl apply -f samples/addons/kiali.yaml
kubectl apply -f samples/addons/prometheus.yaml
kubectl rollout status deployment/kiali -n istio-system

# Enable automatic sidecar injection for kube-goat namespaces
kubectl label namespace big-monolith istio-injection=enabled
kubectl label namespace secure-middleware istio-injection=enabled
kubectl label namespace default istio-injection=enabled

# Restart pods so sidecars get injected
kubectl rollout restart deployment -n big-monolith
kubectl rollout restart deployment -n secure-middleware
kubectl rollout restart deployment -n default

# Verify sidecars are running (should show 2/2 or 3/3 READY)
kubectl get pods -n big-monolith

# Apply mTLS STRICT mode for all kube-goat namespaces
kubectl apply -f istio/mtls/peer-authentication.yaml

# Verify mTLS policies
kubectl get peerauthentication -A

# Access Kiali dashboard
kubectl port-forward svc/kiali -n istio-system 20001:20001
# → http://localhost:20001
# Graph → Display → Security → lock icons = mTLS enforced

# Generate traffic for Kiali visualization
for i in $(seq 1 20); do
  kubectl exec -n big-monolith \
    $(kubectl get pod -n big-monolith -o jsonpath='{.items[0].metadata.name}') \
    -c hunger-check -- wget -q -O- \
    http://metadata-db.default.svc.cluster.local 2>/dev/null || true
  sleep 2
done

ATTACK SCENARIOS — PoC Triggers
----------------------------------------------------------------
# See falco/Attacks PoC.txt for full attack documentation
# Each PoC corresponds to a screenshot in falco/

# Set target pod variable
TARGET=$(kubectl get pod -n big-monolith -o jsonpath='{.items[0].metadata.name}')

# PoC 1 — Sensitive File Access (T1555)
# Evidence: falco-attack2-sensitive-file-shadow.png
kubectl exec -n big-monolith $TARGET -- cat /etc/shadow
# Detected by: Read sensitive file untrusted (Warning)

# PoC 2 — Terminal Shell in Container (T1059)
kubectl exec -it -n big-monolith $TARGET -- /bin/bash
# Detected by: Terminal shell in container (Notice)

# PoC 3 — Fileless Malware via /dev/shm (T1620)
# Evidence: falco-attack1-devshm-execution.png
kubectl exec -n big-monolith $TARGET -- \
  sh -c "cp /bin/sh /dev/shm/evil && /dev/shm/evil -c id"
# Detected by: Execution from /dev/shm (Critical)

# PoC 4 — Service Account Token Access (T1528)
# Evidence: falco-attack3-serviceaccount-token.png
kubectl exec -n big-monolith $TARGET -- \
  cat /var/run/secrets/kubernetes.io/serviceaccount/token
# Detected by: Read sensitive file untrusted (Warning)

# PoC 5 — C2 Outbound Connection (T1105)
# Evidence: falco-attack4-curl-c2.png
kubectl exec -n big-monolith $TARGET -- \
  curl -s http://example.com
# Detected by: Network tool execution in container (Warning)

# PoC 6 — K8s API Lateral Movement (T1613)
# Evidence: falco-attack5-k8s-api-lateral-movement.png
kubectl exec -n big-monolith $TARGET -- \
  curl -k https://kubernetes.default.svc/api/v1/namespaces
